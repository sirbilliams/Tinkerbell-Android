cmake_minimum_required(VERSION 3.16)
project(tinkerbell CXX C)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# Desktop: SDL2 from the system. Android: android/app/jni/CMakeLists.txt adds SDL first, then this file.
option(TB_SANITIZE "Build with ASan + UBSan (run before every APK: Android's allocator crashes on use-after-free)" OFF)

if(NOT TARGET SDL2::SDL2 AND NOT TARGET SDL2 AND NOT TARGET SDL2::SDL2-static)
  find_package(SDL2 REQUIRED)
endif()
if(TARGET SDL2::SDL2)
  set(TB_SDL SDL2::SDL2)
elseif(TARGET SDL2)
  set(TB_SDL SDL2)
elseif(TARGET SDL2::SDL2-static)
  set(TB_SDL SDL2::SDL2-static)
else()
  set(TB_SDL ${SDL2_LIBRARIES})
endif()

# platform/ = the Win32/DirectX shim on SDL2; game/src = the reconstructed game (Vision engine + game classes).
file(GLOB_RECURSE TB_SRC CONFIGURE_DEPENDS platform/*.cpp game/src/*.cpp)

if(ANDROID)
  add_library(tinkerbell SHARED ${TB_SRC})
  target_link_libraries(tinkerbell PRIVATE ${TB_SDL} log android dl m)
else()
  add_executable(tinkerbell ${TB_SRC})
  if(TARGET SDL2::SDL2main)
    target_link_libraries(tinkerbell PRIVATE SDL2::SDL2main)
  endif()
  target_link_libraries(tinkerbell PRIVATE ${TB_SDL} dl m)
endif()
target_include_directories(tinkerbell PRIVATE platform game/src ${SDL2_INCLUDE_DIRS})
target_compile_options(tinkerbell PRIVATE -Wall -Wno-unused-variable -Wno-unused-function -fno-strict-aliasing -fwrapv)
if(TB_SANITIZE)
  target_compile_options(tinkerbell PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(tinkerbell PRIVATE -fsanitize=address,undefined)
endif()
